在 Python 里，标识符由字母、数字、下划线组成。

在 Python 中，所有标识符可以包括英文、数字以及下划线(_)，但不能以数字开头。

# daydayup.py
def dayUp(df):
    dayup = 1
    for i in range(365):
        if i % 7 in[6,0]:
            dayup=dayup*(0.99)
        else:
            dayup =dayup*(1+df)
    return dayup
dayfactor = 0.01
while dayUp(dayfactor) < 37.78:
    dayfactor+=0.001
print("他工作日努力参数为{:.3f}".format(dayfactor))
